{{/* image-optimize: resolve image resource and output tag. Params (dict): page — page context for .Resources.GetMatch src — image source path string alt — alt text fill — (optional) Fill spec, e.g. "800x300 webp q80" process — (optional) Process spec, e.g. "webp q85" If neither fill nor process is given, outputs the raw image. */}} {{- $page := .page -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $img := "" -}} {{- with $page.Resources.GetMatch $src -}} {{- $img = . -}} {{- end -}} {{- if not $img -}} {{- with resources.Get ($src | strings.TrimPrefix "/") -}} {{- $img = . -}} {{- end -}} {{- end -}} {{- if $img -}} {{- if .fill -}} {{- $processed := $img.Fill .fill -}} {{- else if .process -}} {{- $processed := $img.Process .process -}} {{- else -}} {{- end -}} {{- else -}} {{- end -}}
